### 4.4 可信度算法（W2-11 ~ W2-15）

```typescript
// D-S 证据理论核心
export interface MassFunction {
  readonly sourceId: string;
  readonly sourceName: string;
  readonly mass: Map<Set<string>, number>;  // focal elements → mass
  readonly confidence: number;
}

// PCR5 融合规则
export function pcr5(m1: MassFunction, m2: MassFunction): MassFunction {
  // 1. 计算冲突 K
  const K = computeConflict(m1, m2);
  if (K < 0.3) {
    // 低冲突：用 Dempster 规则
    return dempsterCombine(m1, m2);
  }
  // 高冲突：用 PCR5
  return pcr5Combine(m1, m2);
}
```
